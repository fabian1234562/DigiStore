   407→    case 'oferta': return 'bg-red-100 text-red-700 border-red-200';

   416→export function ProductDetail() {
   417→  const { selectedProduct, productDetailOpen, setProductDetailOpen, setSelectedProduct, addToCart, setCartOpen } = useStore();
   418→
   419→  if (!selectedProduct) return null;
   420→
   421→  const product = selectedProduct;
   422→  const delivery = getDeliveryInfo(product);
   423→  const features = getProductFeatures(product);
   424→  const discount = product.originalPrice
   425→    ? Math.round(((product.originalPrice - product.price) / product.originalPrice) * 100)
   426→    : 0;
   427→  const savings = product.originalPrice ? (product.originalPrice - product.price).toFixed(2) : null;
   428→  const DeliveryIcon = delivery.icon;
   429→
   430→  const handleAddToCart = () => {
   431→    addToCart(product);
   432→    setProductDetailOpen(false);
