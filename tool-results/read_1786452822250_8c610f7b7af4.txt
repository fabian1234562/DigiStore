    56→export const useStore = create<StoreState>((set, get) => ({
    57→  cart: [],
    58→  cartOpen: false,
    59→  authOpen: false,
    60→  selectedProduct: null,
    61→  productDetailOpen: false,
    62→  user: null,
    63→  searchQuery: '',
    64→  selectedCategory: 'all',
    65→  selectedSubcategory: 'all',
    66→  sortBy: 'popular',
    67→
    68→  addToCart: (product) =>
    69→    set((state) => {
    70→      const existing = state.cart.find((item) => item.product.id === product.id);
    71→      if (existing) {
    72→        return {
    73→          cart: state.cart.map((item) =>
    74→            item.product.id === product.id
    75→              ? { ...item, quantity: item.quantity + 1 }
    76→              : item
    77→          ),
    78→        };
    79→      }
    80→      return { cart: [...state.cart, { product, quantity: 1 }] };
    81→    }),
    82→
    83→  removeFromCart: (productId) =>
    84→    set((state) => ({
    85→      cart: state.cart.filter((item) => item.product.id !== productId),
    86→    })),
    87→
    88→  updateQuantity: (productId, quantity) =>
    89→    set((state) => ({
    90→      cart:
    91→        quantity <= 0
    92→          ? state.cart.filter((item) => item.product.id !== productId)
    93→          : state.cart.map((item) =>
    94→              item.product.id === productId ? { ...item, quantity } : item
    95→            ),
    96→    })),
    97→
    98→  clearCart: () => set({ cart: [] }),
    99→  setCartOpen: (open) => set({ cartOpen: open }),
   100→  setAuthOpen: (open) => set({ authOpen: open }),
   101→  setSelectedProduct: (product) => set({ selectedProduct: product }),
   102→  setProductDetailOpen: (open) => set({ productDetailOpen: open }),
   103→  setUser: (user) => set({ user }),
   104→  setSearchQuery: (query) => set({ searchQuery: query }),
   105→  setSelectedCategory: (category) => set({ selectedCategory: category, selectedSubcategory: 'all' }),
   106→  setSelectedSubcategory: (subcategory) => set({ selectedSubcategory: subcategory }),
   107→  setSortBy: (sort) => set({ sortBy: sort }),
   108→
   109→  cartTotal: () =>
   110→    get().cart.reduce((total, item) => total + item.product.price * item.quantity, 0),
   111→
   112→  cartCount: () =>
   113→    get().cart.reduce((count, item) => count + item.quantity, 0),
   114→}));
